# Opt in: the transport-only SDK/build remains usable without Vulkan headers.
option(VR_BUILD_VULKAN_MOTION "Build RK3588 Vulkan ROI motion magnifier" OFF)
if(VR_BUILD_VULKAN_MOTION)
    find_package(Vulkan REQUIRED)
    find_program(VR_GLSLANG_VALIDATOR glslangValidator)
    if(NOT VR_GLSLANG_VALIDATOR)
        message(FATAL_ERROR "Host glslangValidator is required for Vulkan shaders")
    endif()
    include(GNUInstallDirs)
    set(VR_MOTION_SPV "${CMAKE_CURRENT_BINARY_DIR}/motion.comp.spv")
    add_custom_command(OUTPUT "${VR_MOTION_SPV}"
        COMMAND "${VR_GLSLANG_VALIDATOR}" -V --target-env vulkan1.0
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders/motion.comp" -o "${VR_MOTION_SPV}"
        DEPENDS shaders/motion.comp VERBATIM)
    add_custom_target(vr-motion-shader DEPENDS "${VR_MOTION_SPV}")
    add_library(vr-vulkan-motion src/vulkan_motion.cpp src/offline_motion.cpp)
    target_compile_features(vr-vulkan-motion PUBLIC cxx_std_17)
    target_include_directories(vr-vulkan-motion PUBLIC src)
    target_link_libraries(vr-vulkan-motion PUBLIC Vulkan::Vulkan)
    target_compile_options(vr-vulkan-motion PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
    add_executable(vr-motion src/motion_main.cpp src/camera.cpp)
    target_link_libraries(vr-motion PRIVATE vr-vulkan-motion Threads::Threads)
    target_compile_options(vr-motion PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(vr-motion PRIVATE
        VR_MOTION_SHADER_PATH="${CMAKE_INSTALL_FULL_DATADIR}/vibrareveal/motion.comp.spv")
    add_dependencies(vr-motion vr-motion-shader)
    install(TARGETS vr-motion RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(TARGETS vr-vulkan-motion ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES src/vulkan_motion.hpp src/offline_motion.hpp src/frame.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/vibrareveal)
    install(FILES "${VR_MOTION_SPV}" DESTINATION ${CMAKE_INSTALL_DATADIR}/vibrareveal)
endif()

if(VR_BUILD_VULKAN_MOTION)

if(VR_BUILD_TOUCH_UI)
    # Target and host tools must describe the actual board runtime. Do not relax
    # this to Ubuntu's 5.15.3 or rewrite Qt version files to pass the gate.
    find_package(Qt5 5.15.10 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
    foreach(qt_host_tool moc rcc)
        get_target_property(qt_host_executable Qt5::${qt_host_tool} IMPORTED_LOCATION)
        execute_process(COMMAND "${qt_host_executable}" -v RESULT_VARIABLE qt_host_result
            OUTPUT_VARIABLE qt_host_stdout ERROR_VARIABLE qt_host_stderr)
        if(NOT qt_host_result EQUAL 0 OR NOT "${qt_host_stdout}${qt_host_stderr}" MATCHES "5\\.15\\.10([^0-9]|$)")
            message(FATAL_ERROR "Qt host ${qt_host_tool} must be runnable version 5.15.10: ${qt_host_executable}")
        endif()
    endforeach()
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTORCC ON)
    add_library(vr-touch-ui src/gui/engine.cpp src/gui/controller.cpp src/gui/frame_view.cpp src/gui/focus_peaking.cpp src/gui/sensor_controls.cpp src/camera.cpp)
    target_include_directories(vr-touch-ui PUBLIC src/gui)
    target_link_libraries(vr-touch-ui PUBLIC vr-touch-core vr-vulkan-motion
        Qt5::Core Qt5::Gui Qt5::Qml Qt5::Quick Qt5::QuickControls2)
    target_compile_options(vr-touch-ui PRIVATE -Wall -Wextra -Wpedantic)
    if(VR_BUILD_RKNN_SELECTION)
        target_link_libraries(vr-touch-ui PRIVATE vr-mobile-sam)
        target_compile_definitions(vr-touch-ui PUBLIC VR_TOUCH_RKNN=1)
    endif()
    add_executable(vr-touch src/gui/main.cpp qml/resources.qrc)
    target_link_libraries(vr-touch PRIVATE vr-touch-ui)
    target_compile_definitions(vr-touch PRIVATE VR_MOTION_SHADER_PATH="${CMAKE_INSTALL_FULL_DATADIR}/vibrareveal-touch/motion.comp.spv")
    add_dependencies(vr-touch vr-motion-shader)
    install(TARGETS vr-touch RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES "${VR_MOTION_SPV}" DESTINATION ${CMAKE_INSTALL_DATADIR}/vibrareveal-touch)
    if(BUILD_TESTING)
        find_package(Qt5 5.15.10 EXACT REQUIRED COMPONENTS Test)
        add_executable(vr-touch-ui-test tests/gui/ui_test.cpp qml/resources.qrc)
        target_link_libraries(vr-touch-ui-test PRIVATE vr-touch-ui Qt5::Test)
        target_compile_definitions(vr-touch-ui-test PRIVATE VR_TEST_SHADER="${VR_MOTION_SPV}")
        add_dependencies(vr-touch-ui-test vr-motion-shader)
        add_test(NAME touch-ui COMMAND vr-touch-ui-test)
        set_tests_properties(touch-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 90)
    endif()
endif()
