cmake_minimum_required(VERSION 3.16)
project(vr_stream LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(vr-stream src/main.cpp src/camera.cpp src/bulk_transport.cpp)

# Opt in: the transport-only SDK/build remains usable without Vulkan headers.
option(VR_BUILD_VULKAN_MOTION "Build RK3588 Vulkan ROI motion magnifier" OFF)
if(VR_BUILD_VULKAN_MOTION)
    find_package(Vulkan REQUIRED)
    find_program(VR_GLSLANG_VALIDATOR glslangValidator)
    if(NOT VR_GLSLANG_VALIDATOR)
        message(FATAL_ERROR "Host glslangValidator is required for Vulkan shaders")
    endif()
    include(GNUInstallDirs)
    set(VR_MOTION_SPV "${CMAKE_CURRENT_BINARY_DIR}/motion.comp.spv")
    add_custom_command(OUTPUT "${VR_MOTION_SPV}"
        COMMAND "${VR_GLSLANG_VALIDATOR}" -V --target-env vulkan1.0
            "${CMAKE_CURRENT_SOURCE_DIR}/shaders/motion.comp" -o "${VR_MOTION_SPV}"
        DEPENDS shaders/motion.comp VERBATIM)
    add_custom_target(vr-motion-shader DEPENDS "${VR_MOTION_SPV}")
    add_library(vr-vulkan-motion src/vulkan_motion.cpp src/offline_motion.cpp)
    target_compile_features(vr-vulkan-motion PUBLIC cxx_std_17)
    target_include_directories(vr-vulkan-motion PUBLIC src)
    target_link_libraries(vr-vulkan-motion PUBLIC Vulkan::Vulkan)
    target_compile_options(vr-vulkan-motion PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
    add_executable(vr-motion src/motion_main.cpp src/camera.cpp)
    target_link_libraries(vr-motion PRIVATE vr-vulkan-motion Threads::Threads)
    target_compile_options(vr-motion PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(vr-motion PRIVATE
        VR_MOTION_SHADER_PATH="${CMAKE_INSTALL_FULL_DATADIR}/vibrareveal/motion.comp.spv")
    add_dependencies(vr-motion vr-motion-shader)
    install(TARGETS vr-motion RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(TARGETS vr-vulkan-motion ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES src/vulkan_motion.hpp src/offline_motion.hpp src/frame.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/vibrareveal)
    install(FILES "${VR_MOTION_SPV}" DESTINATION ${CMAKE_INSTALL_DATADIR}/vibrareveal)
endif()

if(VR_BUILD_VULKAN_MOTION)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(vr-motion-test tests/motion_test.cpp)
        target_link_libraries(vr-motion-test PRIVATE vr-vulkan-motion)

endif()

# Runtime is loaded on demand. Building selection does not require an NPU or
# vendor binary on the build host; the SDK header must be supplied explicitly.
option(VR_BUILD_RKNN_SELECTION "Build MobileSAM object-selection backend" OFF)
if(VR_BUILD_RKNN_SELECTION)
    find_path(RKNN_INCLUDE_DIR rknn_api.h)
    if(NOT RKNN_INCLUDE_DIR)
        message(FATAL_ERROR "Set RKNN_INCLUDE_DIR to the Rockchip SDK include directory")
    endif()
    add_library(vr-mobile-sam src/mobile_sam.cpp)
    target_compile_features(vr-mobile-sam PUBLIC cxx_std_17)
    target_include_directories(vr-mobile-sam PUBLIC src PRIVATE "${RKNN_INCLUDE_DIR}")
    target_link_libraries(vr-mobile-sam PRIVATE ${CMAKE_DL_LIBS})
    target_compile_options(vr-mobile-sam PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(vr-select src/select_main.cpp)
    target_link_libraries(vr-select PRIVATE vr-mobile-sam)
    install(TARGETS vr-select RUNTIME DESTINATION bin)
    install(TARGETS vr-mobile-sam ARCHIVE DESTINATION lib)
    install(FILES src/mobile_sam.hpp src/vulkan_motion.hpp src/frame.hpp DESTINATION include/vibrareveal)
endif()
