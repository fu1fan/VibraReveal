cmake_minimum_required(VERSION 3.16)
project(vr_stream LANGUAGES CXX)
find_package(Threads REQUIRED)
add_executable(vr-stream src/main.cpp src/camera.cpp src/bulk_transport.cpp)
target_compile_features(vr-stream PRIVATE cxx_std_17)
target_compile_options(vr-stream PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(vr-stream PRIVATE Threads::Threads)
install(TARGETS vr-stream RUNTIME DESTINATION bin)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(vr-bulk-source src/bulk_bench.cpp)
    target_compile_features(vr-bulk-source PRIVATE cxx_std_17)
    target_compile_options(vr-bulk-source PRIVATE -Wall -Wextra -Wpedantic)
endif()
