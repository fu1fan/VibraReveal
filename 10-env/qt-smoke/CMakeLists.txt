cmake_minimum_required(VERSION 3.16)
project(elf2_qt_smoke LANGUAGES CXX)
if(CMAKE_CROSSCOMPILING)
    find_package(Qt5 5.15.10 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
else()
    find_package(Qt5 5.15.10 EXACT REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)
endif()
if(NOT CMAKE_CROSSCOMPILING)
    find_package(Qt5 5.15.10 EXACT REQUIRED COMPONENTS Test QuickTest)
endif()
set(CMAKE_AUTOMOC ON)
add_executable(qt-smoke main.cpp)
target_compile_features(qt-smoke PRIVATE cxx_std_17)
target_link_libraries(qt-smoke PRIVATE Qt5::Core Qt5::Gui Qt5::Qml Qt5::Quick Qt5::QuickControls2)
